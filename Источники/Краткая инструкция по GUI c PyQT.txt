Полный код модуля:
import sys
from PyQt6 import uic
from PyQt6.QtWidgets import QApplication
from PyQt6.QtSql import *
from PyQt6.QtCore import Qt


Form, Window = uic.loadUiType("mm.ui")

db_file = 'F_usd3.db'


def test_click():
    print("test ok")


def connect_db(db_file):
    db = QSqlDatabase.addDatabase("QSQLITE")
    db.setDatabaseName(db_file)
    if not db.open():
        print("Cannot establish a database connection to {}!".format(db_file))
        return False
    return db


app = QApplication([])
window = Window()
form = Form()
form.setupUi(window)

if not connect_db(db_file):
    sys.exit(-1)
else:
    print("connection ok")


form.btnTest.clicked.connect(test_click)

model = QSqlTableModel()
model.setTable('F_usd')
model.select()

form.tableView.setSortingEnabled(True)

form.tableView.setModel(model)

window.show()
app.exec()
